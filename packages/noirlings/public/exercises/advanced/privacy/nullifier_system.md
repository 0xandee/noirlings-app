              }
          }

#[test]
fn test_nullifier_generation() {
    let voter = Voter {
        voter_id: 12345,
        secret: 67890,
        nullifier_secret: 11111
    };

    let election_id = 1;
    let nullifier1 = voter.generate_nullifier(election_id);
    let nullifier2 = voter.generate_nullifier(election_id);

    // Same voter, same election should produce same nullifier
    assert(nullifier1 == nullifier2);

    // Different election should produce different nullifier
    let different_election = 2;
    let nullifier3 = voter.generate_nullifier(different_election);
    assert(nullifier1 != nullifier3);

    // Voter should be able to verify their own nullifier
    assert(voter.verify_nullifier(nullifier1, election_id));
    assert(!voter.verify_nullifier(nullifier3, election_id));
}

fn test_cross_election_nullifier_isolation() {

        voter_id: 1,
        secret: 1111,
        nullifier_secret: 2222

    // Create trackers for different elections
    let mut tracker1 = NullifierTracker::new(1);
    let mut tracker2 = NullifierTracker::new(2);

    // Generate nullifiers for both elections
    let nullifier1 = voter.generate_nullifier(1);
    let nullifier2 = voter.generate_nullifier(2);
    let proof1 = voter.generate_nullifier_proof(1);
    let proof2 = voter.generate_nullifier_proof(2);

    // Should be able to vote in both elections
    assert(tracker1.record_nullifier(nullifier1, proof1));
    assert(tracker2.record_nullifier(nullifier2, proof2));

    // Each tracker should only know about its own nullifier
    assert(tracker1.is_nullifier_used(nullifier1));
    assert(!tracker1.is_nullifier_used(nullifier2));
    assert(tracker2.is_nullifier_used(nullifier2));
    assert(!tracker2.is_nullifier_used(nullifier1));
}
